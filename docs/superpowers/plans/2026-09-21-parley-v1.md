# Parley v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Parley, a macOS menu bar app where a tap-tap-hold gesture on a modifier key records speech, transcribes it with Groq Whisper and pastes the text at the cursor.

**Architecture:** The Electron main process owns all state. Small units under `electron/` (gesture detector, key hook, transcriber, injector, settings store, session) take their dependencies as arguments, so they are unit-tested without importing Electron. Two extra renderer entries are added next to the scaffold's demo app: `orb.html` (microphone capture plus the orb animation) and `settings.html` (settings and onboarding).

**Tech Stack:** Electron 30, Vite 5 with `vite-plugin-electron`, React 19, Mantine 9, TanStack Query, React Hook Form + Zod, Vitest, `uiohook-napi` (global key hook and paste keystroke), Groq `whisper-large-v3-turbo`.

**Spec:** `docs/superpowers/specs/2026-09-21-parley-design.md`

## Global Constraints

- Target: macOS 26, Apple Silicon (arm64). Personal use, runs from a local build. No signing or notarization.
- Groq endpoint `https://api.groq.com/openai/v1/audio/transcriptions`, `model=whisper-large-v3-turbo`, `language=en`, vocabulary sent as `prompt`.
- Raw Whisper output is pasted as is. No LLM cleanup.
- Gesture timing: tap shorter than 250 ms, gap under 300 ms, final press held longer than 150 ms. Tap count 2 to 4, default 3. Default key Right Option. Modifier keys only.
- Key events are observed, never swallowed.
- Audio shorter than 300 ms is discarded. Recording auto-stops at 10 minutes.
- Orb timing: done flash 300 ms, error 3 s. The orb window is never focusable and is created once, then hidden, never destroyed.
- On any error nothing is pasted and the clipboard is untouched.
- The API key is stored only through Electron `safeStorage` and never sent to a renderer.
- Harness rules from `AGENTS.md` apply: exactly one `in_progress` feature in `feature_list.json`; the renderer reaches the OS only through `electron/preload.ts`; no raw hex or `rgb()` colours and no utility classes in `src/`; Mantine components instead of raw `<button>`, `<input>`, `<textarea>`, `<select>`; every new dependency justified in `DECISIONS.md`.
- `make check` must exit 0 at the end of every task.
- All Vitest files run in the scaffold's global `jsdom` environment with `src/test/setup.ts`. Do not add `@vitest-environment node` (the setup file touches `window`).
- The scaffold's demo app (`index.html`, `src/pages`, `src/components`) is left untouched and is simply not opened by Parley. Removing it is a separate cleanup.
- Every commit message ends with the attribution lines the session provides.

```ts
import type { GestureKey, KeyEvent, TapCount } from './types';

/**
 * Tap-tap-hold, as a pure state machine.
 *
 * No Electron imports, so the whole gesture is covered by unit tests. It only
 * observes events; nothing here can swallow a key.
 */

export const TAP_MAX_MS = 250;
export const GAP_MAX_MS = 300;
export const HOLD_MIN_MS = 150;

export interface GestureDetectorOptions {
  key: GestureKey;
  tapCount: TapCount;
  onStart(): void;
  onStop(): void;
  onCancel(): void;
}

export interface GestureDetector {
  handle(event: KeyEvent): void;
  configure(config: { key: GestureKey; tapCount: TapCount }): void;
}

export function createGestureDetector(options: GestureDetectorOptions): GestureDetector {
  let key = options.key;
  let tapCount = options.tapCount;

  let taps = 0;
  let downAt: number | null = null;
  let lastUpAt: number | null = null;
  let holdTimer: ReturnType<typeof setTimeout> | undefined;
  let recording = false;

  function reset() {
    taps = 0;
    downAt = null;
    lastUpAt = null;
    if (holdTimer) clearTimeout(holdTimer);
    holdTimer = undefined;
  }

  function handle(event: KeyEvent) {
    if (recording) {
      if (event.key === 'Escape' && event.type === 'down') {
        recording = false;
        reset();
        options.onCancel();
      } else if (event.key === key && event.type === 'up') {
        recording = false;
        reset();
        options.onStop();
      }
      return;
    }

    if (event.key !== key) {
      // Option+letter and friends must never look like a tap.
      if (event.type === 'down') reset();
      return;
    }

    if (event.type === 'down') {
      if (downAt !== null) return; // auto-repeat
      if (lastUpAt !== null && event.time - lastUpAt > GAP_MAX_MS) reset();
      downAt = event.time;
      if (taps === tapCount - 1) {
        holdTimer = setTimeout(() => {
          holdTimer = undefined;
          recording = true;
          options.onStart();
        }, HOLD_MIN_MS);
      }
      return;
    }

    if (downAt === null) return;
    const heldFor = event.time - downAt;
    downAt = null;

    // Released the final press before it became a hold, or held a tap too long.
    if (holdTimer || heldFor >= TAP_MAX_MS) {
      reset();
      return;
    }

    taps += 1;
    lastUpAt = event.time;
  }

  function configure(config: { key: GestureKey; tapCount: TapCount }) {
    key = config.key;
    tapCount = config.tapCount;
    recording = false;
    reset();
  }

  return { handle, configure };
}
```

```ts
import { GESTURE_KEYS, type KeyEvent, type KeyName } from './types';

/**
 * Adapter between `uiohook-napi` and the gesture detector. The hook and its
 * keycode table are passed in, so this file never imports the native module
 * and can be tested anywhere.
 */

export interface HookLike {
  on(event: 'keydown' | 'keyup', listener: (e: { keycode: number }) => void): unknown;
  start(): void;
  stop(): void;
}

export function createKeyHook(
  hook: HookLike,
  keycodes: Record<string, number>,
  onEvent: (event: KeyEvent) => void,
  now: () => number = () => performance.now()
) {
  const names = new Map<number, KeyName>();
  for (const name of [...GESTURE_KEYS, 'Escape'] as const) {
    const code = keycodes[name];
    if (typeof code === 'number') names.set(code, name);
  }

  const forward = (type: KeyEvent['type']) => (e: { keycode: number }) =>
```

```ts
import type { KeyTestResult } from './types';

/**
 * The only file that talks to Groq. If a cleanup pass is ever added, it wraps
 * `transcribe` here and nothing else changes.
 */

const BASE = 'https://api.groq.com/openai/v1';
const MODEL = 'whisper-large-v3-turbo';

export type TranscribeErrorKind = 'no_key' | 'network' | 'auth' | 'rate_limit' | 'server';

const MESSAGES: Record<TranscribeErrorKind, string> = {
  no_key: 'No API key',
  network: 'No connection',
  auth: 'Invalid API key',
  rate_limit: 'Rate limited',
  server: 'Groq error',
};

export class TranscribeError extends Error {
  constructor(public readonly kind: TranscribeErrorKind) {
    super(MESSAGES[kind]);
    this.name = 'TranscribeError';
  }
}

function kindForStatus(status: number): TranscribeErrorKind {
  if (status === 401 || status === 403) return 'auth';
  if (status === 429) return 'rate_limit';
  return 'server';
}

export interface Transcriber {
  transcribe(
    audio: { data: ArrayBuffer; mimeType: string },
    options: { vocabulary: string }
  ): Promise<string>;
  testKey(key: string): Promise<KeyTestResult>;
}

export function createTranscriber(deps: {
  getApiKey(): string | null;
  fetchFn?: typeof fetch;
}): Transcriber {
  const fetchFn = deps.fetchFn ?? fetch;

  async function request(url: string, init: RequestInit): Promise<Response> {
    let response: Response;
    try {
      response = await fetchFn(url, init);
    } catch {
      throw new TranscribeError('network');
    }
    if (!response.ok) throw new TranscribeError(kindForStatus(response.status));
    return response;
  }

  return {
    async transcribe(audio, { vocabulary }) {
      const key = deps.getApiKey();
      if (!key) throw new TranscribeError('no_key');

      const form = new FormData();
      form.set('file', new Blob([audio.data], { type: audio.mimeType }), 'speech.webm');
      form.set('model', MODEL);
      form.set('language', 'en');
      form.set('response_format', 'json');
      if (vocabulary.trim()) form.set('prompt', vocabulary.trim());

      const response = await request(`${BASE}/audio/transcriptions`, {
        method: 'POST',
        headers: { Authorization: `Bearer ${key}` },
        body: form,
      });
      const body = (await response.json()) as { text?: string };
      return (body.text ?? '').trim();
    },

    async testKey(key) {
      try {
        await request(`${BASE}/models`, {
          method: 'GET',
          headers: { Authorization: `Bearer ${key}` },
        });
        return { ok: true, message: 'Key works' };
      } catch (error) {
        return { ok: false, message: (error as Error).message };
      }
    },
  };
}
```

```ts
/**
 * Paste at the cursor by borrowing the clipboard: snapshot it, write the
 * transcript, send Cmd+V, then put the snapshot back. The restore runs in a
 * `finally`, so a failed paste never costs the user their clipboard.
 */

/** Let the pasteboard settle before the keystroke reads it. */
export const PASTE_SETTLE_MS = 50;
/** Let the target app read the pasteboard before it is restored. */
export const RESTORE_DELAY_MS = 200;

export interface InjectorDeps<Snapshot> {
  snapshot(): Snapshot;
  restore(snapshot: Snapshot): void;
  writeText(text: string): void;
  sendPaste(): void;
  delay?(ms: number): Promise<void>;
}

export interface Injector {
  paste(text: string): Promise<void>;
}

const sleep = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms));

export function createInjector<Snapshot>(deps: InjectorDeps<Snapshot>): Injector {
  const delay = deps.delay ?? sleep;

  return {
    async paste(text) {
      const saved = deps.snapshot();
      try {
        deps.writeText(text);
        await delay(PASTE_SETTLE_MS);
        deps.sendPaste();
        await delay(RESTORE_DELAY_MS);
      } finally {
        deps.restore(saved);
      }
    },
  };
}
```

```ts
import { readFileSync, writeFileSync } from 'node:fs';
import path from 'node:path';

import { DEFAULT_SETTINGS, GESTURE_KEYS, type Settings, type SettingsView } from './types';

/** The shape of Electron's `safeStorage`, passed in so tests need no Electron. */
export interface CryptoLike {
  isEncryptionAvailable(): boolean;
  encryptString(plain: string): Buffer;
  decryptString(encrypted: Buffer): string;
}

export interface SettingsStore {
  get(): Settings;
  update(patch: Partial<Settings>): Settings;
  view(): SettingsView;
  getApiKey(): string | null;
  setApiKey(key: string): void;
  hasApiKey(): boolean;
}

interface FileShape {
  settings?: Partial<Settings>;
  apiKeyEncrypted?: string;
}

/** Keep only fields with valid values. The renderer is untrusted input. */
function sanitize(input: Partial<Settings>): Partial<Settings> {
  const out: Partial<Settings> = {};
  if (GESTURE_KEYS.includes(input.gestureKey as never)) out.gestureKey = input.gestureKey;
  if (input.tapCount === 2 || input.tapCount === 3 || input.tapCount === 4) {
    out.tapCount = input.tapCount;
  }
  if (typeof input.vocabulary === 'string') out.vocabulary = input.vocabulary.slice(0, 2000);
  if (typeof input.tickSound === 'boolean') out.tickSound = input.tickSound;
  if (typeof input.launchAtLogin === 'boolean') out.launchAtLogin = input.launchAtLogin;
  return out;
}

export function createSettingsStore(deps: { dir: string; crypto: CryptoLike }): SettingsStore {
  const file = path.join(deps.dir, 'settings.json');

  let state: FileShape = {};
  try {
    state = JSON.parse(readFileSync(file, 'utf8')) as FileShape;
  } catch {
    // First launch or a corrupted file: defaults are fine.
  }

  let settings: Settings = { ...DEFAULT_SETTINGS, ...sanitize(state.settings ?? {}) };
  let apiKeyEncrypted = typeof state.apiKeyEncrypted === 'string' ? state.apiKeyEncrypted : null;

  function save() {
    const body: FileShape = { settings, ...(apiKeyEncrypted ? { apiKeyEncrypted } : {}) };
    writeFileSync(file, JSON.stringify(body, null, 2));
  }

  return {
    get: () => settings,
    update(patch) {
      settings = { ...settings, ...sanitize(patch) };
      save();
      return settings;
    },
    view: () => ({ ...settings, hasApiKey: apiKeyEncrypted !== null }),
    hasApiKey: () => apiKeyEncrypted !== null,
    getApiKey() {
      if (!apiKeyEncrypted) return null;
      try {
        return deps.crypto.decryptString(Buffer.from(apiKeyEncrypted, 'base64'));
      } catch {
        return null;
      }
    },
    setApiKey(key) {
      if (!deps.crypto.isEncryptionAvailable()) throw new Error('Encryption is not available');
      apiKeyEncrypted = deps.crypto.encryptString(key).toString('base64');
      save();
    },
  };
}
```

```ts
import type { Injector } from './injector';
import type { Transcriber } from './transcriber';
import type { OrbPhasePayload, PermissionStatus, RecordedAudio, Settings } from './types';

/**
 * One dictation session, from the held press to the paste.
 *
 * Everything it touches is passed in, so the rules that matter — the busy
 * guard, the short-audio discard, "an error never touches the clipboard" — are
 * unit tests rather than things to try by hand.
 */

export const MIN_AUDIO_MS = 300;
export const MAX_RECORDING_MS = 600_000;
export const DONE_MS = 300;
export const ERROR_MS = 3000;
export const FADE_MS = 150;

export type SessionState = 'idle' | 'listening' | 'awaiting_audio' | 'transcribing' | 'finishing';

export interface SessionDeps {
  orb: { show(): void; hide(): void; setPhase(payload: OrbPhasePayload): void };
  transcriber: Pick<Transcriber, 'transcribe'>;
  injector: Injector;
  getSettings(): Settings;
  hasApiKey(): boolean;
  getPermissions(): PermissionStatus;
  /** The settings window doubles as onboarding. */
  openSettings(): void;
  onTranscript(text: string): void;
}

export interface Session {
  start(): void;
  stop(): void;
  cancel(): void;
  receiveAudio(audio: RecordedAudio): Promise<void>;
  state(): SessionState;
}

export function createSession(deps: SessionDeps): Session {
  let state: SessionState = 'idle';
  let maxTimer: ReturnType<typeof setTimeout> | undefined;

  function clearMaxTimer() {
    if (maxTimer) clearTimeout(maxTimer);
    maxTimer = undefined;
  }

  /** Show a final phase for `visibleMs`, let the orb fade, then hide it. */
  function finish(payload: OrbPhasePayload, visibleMs: number) {
    state = 'finishing';
    deps.orb.setPhase(payload);
    setTimeout(() => {
      deps.orb.hide();
      state = 'idle';
    }, visibleMs + FADE_MS);
  }

  function stop() {
    if (state !== 'listening') return;
    clearMaxTimer();
    state = 'awaiting_audio';
    deps.orb.setPhase({ phase: 'transcribing', tick: deps.getSettings().tickSound });
  }

  return {
    state: () => state,

    start() {
      if (state !== 'idle') return;
      const permissions = deps.getPermissions();
      if (!permissions.microphone || !permissions.accessibility || !deps.hasApiKey()) {
        deps.openSettings();
        return;
      }
      state = 'listening';
      deps.orb.show();
      deps.orb.setPhase({ phase: 'listening' });
      maxTimer = setTimeout(stop, MAX_RECORDING_MS);
    },

    stop,

    cancel() {
      if (state !== 'listening') return;
      clearMaxTimer();
      finish({ phase: 'cancelled' }, 0);
    },

    async receiveAudio(audio) {
      if (state !== 'awaiting_audio') return;
      if (audio.durationMs < MIN_AUDIO_MS) {
        finish({ phase: 'cancelled' }, 0);
        return;
      }

      state = 'transcribing';
      try {
        const text = await deps.transcriber.transcribe(
          { data: audio.data, mimeType: audio.mimeType },
          { vocabulary: deps.getSettings().vocabulary }
        );
        if (!text) {
          finish({ phase: 'cancelled' }, 0);
          return;
        }
        await deps.injector.paste(text);
        deps.onTranscript(text);
        finish({ phase: 'done' }, DONE_MS);
      } catch (error) {
        const message = error instanceof Error ? error.message : 'Something went wrong';
        finish({ phase: 'error', message }, ERROR_MS);
      }
    },
  };
}
```

```ts
/** Speech RMS is small; this gain maps a normal speaking voice to roughly 0.5. */
const GAIN = 4;

/**
 * Voice level from an `AnalyserNode.getByteTimeDomainData` buffer: RMS around
 * the 128 midpoint, scaled and clamped to 0..1.
 */
export function levelFromSamples(samples: Uint8Array): number {
  if (samples.length === 0) return 0;
  let sum = 0;
  for (const sample of samples) {
    const centered = (sample - 128) / 128;
    sum += centered * centered;
  }
  return Math.min(1, Math.sqrt(sum / samples.length) * GAIN);
}
```

```ts
import type { RecordedAudio } from '../../electron/types';

import { levelFromSamples } from './level';

/**
 * Microphone capture. The stream is opened on `start` and every track is
 * stopped on `stop` or `discard`, so the macOS microphone indicator is lit
 * only while the key is held.
 */

const MIME_TYPE = 'audio/webm;codecs=opus';

export interface Recorder {
  start(): Promise<void>;
  /** Resolves with the recording, or null if nothing was being recorded. */
  stop(): Promise<RecordedAudio | null>;
  discard(): void;
  /** Current voice level, 0 to 1. */
  level(): number;
}

export function createRecorder(): Recorder {
  let stream: MediaStream | null = null;
  let mediaRecorder: MediaRecorder | null = null;
  let audioContext: AudioContext | null = null;
  let analyser: AnalyserNode | null = null;
  let samples = new Uint8Array(0);
  let chunks: Blob[] = [];
  let startedAt = 0;

  function release() {
    stream?.getTracks().forEach((track) => track.stop());
    void audioContext?.close();
    stream = null;
    mediaRecorder = null;
    audioContext = null;
    analyser = null;
    chunks = [];
  }

  return {
    async start() {
      release();
      stream = await navigator.mediaDevices.getUserMedia({
        audio: { channelCount: 1, sampleRate: 16000, echoCancellation: false },
      });
      audioContext = new AudioContext();
      analyser = audioContext.createAnalyser();
      analyser.fftSize = 512;
      samples = new Uint8Array(analyser.fftSize);
      audioContext.createMediaStreamSource(stream).connect(analyser);

      mediaRecorder = new MediaRecorder(stream, {
        mimeType: MIME_TYPE,
        audioBitsPerSecond: 32000,
      });
      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) chunks.push(event.data);
      };
      startedAt = performance.now();
      mediaRecorder.start();
    },

    stop() {
      const active = mediaRecorder;
      if (!active || active.state === 'inactive') {
        release();
        return Promise.resolve(null);
      }
      const durationMs = performance.now() - startedAt;
      return new Promise((resolve) => {
        active.onstop = async () => {
          const data = await new Blob(chunks, { type: MIME_TYPE }).arrayBuffer();
          release();
          resolve({ data, mimeType: 'audio/webm', durationMs });
        };
        active.stop();
      });
    },

    discard() {
      if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.onstop = null;
        mediaRecorder.stop();
      }
      release();
    },

    level() {
      if (!analyser) return 0;
      analyser.getByteTimeDomainData(samples);
      return levelFromSamples(samples);
    },
  };
}
```

```ts
/** A soft, short tick when recording stops. Synthesized, so no audio asset ships. */
export function playTick() {
  const context = new AudioContext();
  const oscillator = context.createOscillator();
  const gain = context.createGain();
  const now = context.currentTime;

  oscillator.type = 'sine';
  oscillator.frequency.setValueAtTime(880, now);
  oscillator.frequency.exponentialRampToValueAtTime(660, now + 0.08);
  gain.gain.setValueAtTime(0.0001, now);
  gain.gain.exponentialRampToValueAtTime(0.15, now + 0.005);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.09);

  oscillator.connect(gain).connect(context.destination);
  oscillator.start(now);
  oscillator.stop(now + 0.1);
  oscillator.onended = () => void context.close();
}
```

```tsx
import { useEffect, useRef, useState } from 'react';
import { Stack, Text } from '@mantine/core';

import type { OrbPhase } from '../../electron/types';

import { createRecorder } from './recorder';
import { playTick } from './tick';
import classes from './Orb.module.css';

const CANVAS_SIZE = 180;
const BASE_RADIUS = 34;

/** Colours come from the theme, never a literal (see `make ui-lint`). */
function themeColor(name: string) {
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
}

function draw(
  context: CanvasRenderingContext2D,
  phase: OrbPhase,
  level: number,
  timeMs: number
) {
  const center = CANVAS_SIZE / 2;
  const color = themeColor(
    phase === 'error'
      ? '--mantine-color-red-6'
      : phase === 'done'
        ? '--mantine-color-teal-5'
        : '--mantine-primary-color-filled'
  );

  // Listening swells with the voice; transcribing breathes on a slow sine.
  const pulse =
    phase === 'listening'
      ? level * 30
      : phase === 'transcribing'
        ? 4 + Math.sin(timeMs / 180) * 4
        : 0;
  const radius = BASE_RADIUS + pulse;

  context.clearRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);

  const glow = context.createRadialGradient(center, center, radius * 0.4, center, center, radius * 1.9);
  glow.addColorStop(0, color);
  glow.addColorStop(1, 'transparent');
  context.globalAlpha = 0.35 + level * 0.4;
  context.fillStyle = glow;
  context.beginPath();
  context.arc(center, center, radius * 1.9, 0, Math.PI * 2);
  context.fill();

  context.globalAlpha = 1;
  context.fillStyle = color;
  context.beginPath();
  context.arc(center, center, radius, 0, Math.PI * 2);
  context.fill();
}

export function Orb() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const phaseRef = useRef<OrbPhase>('cancelled');
  const [visible, setVisible] = useState(false);
  const [message, setMessage] = useState('');

  useEffect(() => {
    const recorder = createRecorder();

    const unsubscribe = window.parley.onOrbPhase(async ({ phase, message: text, tick }) => {
      phaseRef.current = phase;
      setMessage(phase === 'error' ? (text ?? '') : '');
      setVisible(phase !== 'cancelled');

      if (phase === 'listening') {
        await recorder.start();
      } else if (phase === 'transcribing') {
        const audio = await recorder.stop();
        if (tick) playTick();
        if (audio) window.parley.sendAudio(audio);
      } else if (phase === 'cancelled') {
        recorder.discard();
      } else if (phase === 'done') {
        // Main hides the window after the flash; start the fade just before.
        setTimeout(() => setVisible(false), 300);
      } else if (phase === 'error') {
        setTimeout(() => setVisible(false), 3000);
      }
    });

    let frame = 0;
    const loop = (timeMs: number) => {
      const context = canvasRef.current?.getContext('2d');
      if (context) draw(context, phaseRef.current, recorder.level(), timeMs);
      frame = requestAnimationFrame(loop);
    };
    frame = requestAnimationFrame(loop);

    return () => {
      unsubscribe();
      cancelAnimationFrame(frame);
      recorder.discard();
    };
  }, []);

  return (
    <Stack className={classes.root} data-visible={visible} align="center" justify="center" gap={0}>
      <canvas ref={canvasRef} width={CANVAS_SIZE} height={CANVAS_SIZE} />
      <Text size="xs" c="red.4" fw={500} h={16}>
        {message}
      </Text>
    </Stack>
  );
}
```

```ts
/**
 * The tray menu as a pure template, so its wording and wiring are testable.
 * `main.ts` hands the result to `Menu.buildFromTemplate`.
 */

export interface TrayState {
  ready: boolean;
  lastTranscript: string | null;
}

export interface TrayActions {
  copyLast(): void;
  openSettings(): void;
  quit(): void;
}

export interface TrayItem {
  label?: string;
  type?: 'separator';
  enabled?: boolean;
  click?: () => void;
}

export function trayMenuTemplate(state: TrayState, actions: TrayActions): TrayItem[] {
  return [
    { label: state.ready ? 'Parley is ready' : 'Setup needed — open Settings', enabled: false },
    { type: 'separator' },
    {
      label: 'Copy last transcript',
      enabled: state.lastTranscript !== null,
      click: actions.copyLast,
    },
    { label: 'Settings…', click: actions.openSettings },
    { type: 'separator' },
    { label: 'Quit Parley', click: actions.quit },
  ];
}
```

```ts
import { shell, systemPreferences } from 'electron';

import type { PermissionKind, PermissionStatus } from './types';

const PANES: Record<PermissionKind, string> = {
  microphone: 'x-apple.systempreferences:com.apple.preference.security?Privacy_Microphone',
  accessibility: 'x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility',
};

export function getPermissions(): PermissionStatus {
  return {
    microphone: systemPreferences.getMediaAccessStatus('microphone') === 'granted',
    accessibility: systemPreferences.isTrustedAccessibilityClient(false),
  };
}

/**
 * Ask macOS for the permission. The microphone has a real prompt the first
 * time. Accessibility has no grant API, and a denied microphone never prompts
 * again, so in both cases the right System Settings pane is opened instead.
 */
export async function requestPermission(kind: PermissionKind): Promise<PermissionStatus> {
  if (kind === 'microphone') {
    const status = systemPreferences.getMediaAccessStatus('microphone');
    if (status === 'not-determined') await systemPreferences.askForMediaAccess('microphone');
    else if (status !== 'granted') await shell.openExternal(PANES.microphone);
  } else if (!systemPreferences.isTrustedAccessibilityClient(true)) {
    await shell.openExternal(PANES.accessibility);
  }
  return getPermissions();
}
```

```ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';

import type { PermissionKind, Settings } from '../../electron/types';

/**
 * Query hooks over `window.parley`. IPC is this window's "server", so it gets
 * the same treatment as server data: no bridge calls from a component.
 */

const SETTINGS = ['parley', 'settings'] as const;
const PERMISSIONS = ['parley', 'permissions'] as const;

export function useSettings() {
  return useQuery({ queryKey: SETTINGS, queryFn: () => window.parley.getSettings() });
}

export function useUpdateSettings() {
  const client = useQueryClient();
  return useMutation({
    mutationFn: (patch: Partial<Settings>) => window.parley.updateSettings(patch),
    onSuccess: (view) => client.setQueryData(SETTINGS, view),
  });
}

/** Polled, because permissions are granted in System Settings, outside the app. */
export function usePermissions() {
  return useQuery({
    queryKey: PERMISSIONS,
    queryFn: () => window.parley.getPermissions(),
    refetchInterval: 2000,
  });
}

export function useRequestPermission() {
  const client = useQueryClient();
  return useMutation({
    mutationFn: (kind: PermissionKind) => window.parley.requestPermission(kind),
    onSuccess: (status) => client.setQueryData(PERMISSIONS, status),
  });
}

export function useSetApiKey() {
  const client = useQueryClient();
  return useMutation({
    mutationFn: (key: string) => window.parley.setApiKey(key),
    onSuccess: () => client.invalidateQueries({ queryKey: SETTINGS }),
  });
}
```
